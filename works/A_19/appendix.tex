\chapter{Lyrics}

\textit{Unterredende Perſonen.}\\
Jahel, die beherzte Heldin. Canto.\\
Debora, Eine mit Prophetiſchen Geiſt begaabte Richterin in Israel. Alto. \\
Sisera, der Feldfürſt des Cananiter Königs Jabin. Tenore.\\
Barack, der Israelitiſchen Feld-Obriſter. Baſso.\\
Chorus. der ſtreitenden Israelitern.

\section{Actus primus}

\begin{lyrics}{Scena prima}
  \voice[Chorus der Israeliten]
  Gott Israel, blickh auf daß Joch,\\
  ſo unßern Ohnmachtsnaken drüket.\\
  Erbarme dich und helffe doch\\
  eh unßer Hofnungsgeiſt erſtiket.\\
  Zernicht den Zwang von Bänden\\
  der bangen Dienſtbarkeit\\
  und mach von Feundesbanden\\
  dein Israel befreyt,\\
  und mach dein Volk befreyt.
\end{lyrics}

\begin{lyrics}{Scena secunda}
  \voice[Debora]
  Er ſterbe, er ſterbe. Diſer Schluß,\\
  der unßre Freyheit flüglen muß,\\
  ſoll unſrer Feunde Dämpfer ſeyn.

  \voice[Barak]
  Welch Raßen nimmt dich ein?\\
  Laßtu den Sanfftmuethsgeiſt\\
  auß ſeinen Trieben gleitten,\\
  du wüetteſt mit dir ſelbſt,\\
  waß will mir daß bedeuthen?\\
  Wohin verlierſt du dich?

  \voice[Debora]
  Ja Barak höre mich,\\
  und durch mich Gottes Willen,\\
  den Ich und du erfüllen\\
  durch ſein Geheißen ſoll.\\
  Der Herr, von dem daß Wohl\\
  ſowie daß Weh entſprünget,\\
  der Helden höbt und ſtürzt,\\
  der Wueth und Lüſt bezwünget,\\
  der unſern Feund erhub\\
  damit er tieffer fallt,\\
  will nun durch unſer Schwerd,\\
  gelenket durch den Gewald,\\
  der ßeine Allmacht wird\\
  in unſre Kräfften ſenken,\\
  durch den geſtürzten Feund\\
  unß unſer Freyheit ſchenkhen.

  \voice[Barak]
  Wie dunkel redeſt du\\
  von Troſt ergänzten Sachen,\\
  waß ſoll ich mir hierauß\\
  dan vor Begriffe machen.

  \voice[Debora]
  Held, höre fehrner zue,\\
  du wirſt ſogleich daß Liecht\\
  in meinen Reden fünden,\\
  den Werkzeug, welchen Gott\\
  zur Geißel unſrer Sünden\\
  bißhero angewant,\\
  ſoll unſre Tapferkeit\\
  alß zweyter Werkzeug auch\\
  ſelbſt wider brechen,\\
  und zwar in einen Streit,\\
  den du mit Sisera ſolſt wagen,\\
  wo du zwar Israel wirſt rächen,\\
  doch ſeinen Feund nicht ſchlagen.

  \voice[Barak]
  Du ſprichſt geheimnußvoll\\
  und lokerſt meinen Sinn\\
  zu ganz verwirten Denken,\\
  wohin wird endlich ſich\\
  dein klarer Ausſpruch lenken?

  \voice[Debora]
  Ich rede, wie ich ſoll,\\
  nun frage weiter nicht,\\
  die Zeit wird dich ſchon lehren,\\
  waß Sieg und Schlagen heiſt,\\
  jez folge nun dem Herren,\\
  der dich durch beſondern Ruff\\
  zu ſeines Israels\\
  Befreyer hat ernennet,\\
  du ſolſt die Kriegesflamm,\\
  ſo um den Tabor brennet,\\
  mit tapferen Behuf\\
  von zehentauſend Mann\\
  der Kinder Sabulon\\
  und Nephtalim erſtüken.\\
  Doch wird die Siegeskron\\
  von dißesmahl ſich nicht\\
  um deine Schläffe ſchicken,\\
  dan Gott wird Sisera\\
  und ſein erboßtes Leben\\
  in eines Weibes Hände geben.

  \voice[Barak]
  Bringt eines Weibes Hand\\
  ſo villes wohl zu Stand,\\
  daß ſie auch Helden überwindet?

  \voice[Debora]
  Wenn Gott die Siegeskrafft\\
  auf Weiber Armbe bindet,\\
  ſo zwüngt ein Weibe diß,\\
  waß auch bey Helden offt\\
  ſchon unerzwünglich ſcheint und hieß.
\end{lyrics}

\begin{lyrics}{Aria prima}
  \voice[Debora]
  Es muß nicht ſtäts ein Donnerſtein\\
  den ſtolzen Keder ſchmettern,\\
  und nicht alzeit ein Sturmwind ſeyn\\
  die Äeſte abzublättern.\\
  Indeme auch zuweil\\
  ein ſchwach doch ſcharffes Beill\\
  ihn khan zu Boden werffen.\\[1ex]
  Alßo wird ein ſchwaches Weib,\\
  welchen Gott die Hände führet,\\
  wan ſie auch den ſtärkeſten Leib\\
  eines Helden nur berühret\\
  den Sieg erwarthen därffen.
\end{lyrics}

\begin{lyrics}{Scena tertia}
  \voice[Barak]
  Ich ſolle zu dem Streit\\
  und meinen Mueth mit Schweiß und Kräfften parren,
  ja auch mein Blut laßt mich ſo Pflicht,\\
  alß Ruhm nicht vor mich ſelbſt bewahren.\\
  Ich ſolle Führer ſeyn,\\
  Ich ſolle Feunde dämpfen,\\
  Ich ſoll vor Israel,\\
  und ſeine Freyheit kämpfen?\\
  Und dennoch ſollen mir den Lorber,\\
  ſo mein Schwert durch Siegen reif gemacht,\\
  ein ſchwaches Weib abbrechen,\\
  wird diß nicht meinen Ruhm\\
  und daß Vertrauen ſchwächen,\\
  womit mein Kriegesheer\\
  mich liebet und mich auch ehrt.\\
  Nein! Diß ſolle nicht geſchehen,\\
  wan ich den Baum gepflanzt,\\
  ſo will ich auch vor mich allein\\
  die Frücht und Lorber ßehen.\\
  Jedoch was denkhe Ich?\\
  Wohin betaubter Geiſt\\
  von Hochmuth aufgeſchwollen,\\
  kanſtu den Siege dan\\
  auß eignen Kräfften hollen?\\
  Iſts nicht der Helden Gott,\\
  der Siegesreißer ſchenkt,\\
  du biſt der Werkzeug nur,\\
  den da ſein Meiſter lenkt,\\
  wie, und wohin er will,\\
  und komt ſodan ein Wunderwerkh aus Licht,\\
  ſo hat es ja der Werkzeug nicht,\\
  Nein! nur der ſo ihn führt gethan.\\
  Drum forth, du Flügelſinn,\\
  ich gehe meiner Pflicht,\\
  nicht zu dem Hochmueth hin;\\
  und wilſtu einem Weib\\
  ô Gott! den Sieg auch ſchenken,\\
  ſo will ich demuthvoll\\
  auf deinen Willen denkhen.
\end{lyrics}

\begin{lyrics}{Scena quarta}
  \voice[Sisera]
  Wie! wie ſoll der Sclavensgeiſt\\
  der erboßten Israelitern,\\
  anſtat vor unſrer Macht\\
  gebeuget zu erſchittern,\\
  von Hochmuth angefeurt,\\
  der Dienſtbarkeit vergeßen,\\
  die Waffen ihren Herrn\\
  zu zeigen ſich ermeßen.\\
  Verruchter Knechte Schaar!\\
  Waß macht dich dan ſogar\\
  khün frech und völlig loß von Zügel?\\
  Wer ſteuret deinen Stolz\\
  ſo breit geſpannte Flügel?\\
  Alleine fliege nur\\
  und folge deiner Hochmuths Spuhr,\\
  ja troze auf den Siegesnuzen,\\
  mein blankes Schwerdt ſoll dir\\
  die Pfauenfedern ſtuzen.\\
  Ich werd dich auß den Schooß\\
  des dir in frechen Sinn\\
  entworfnen Glüks ſchon dringen,\\
  bevor du dich zu hoch zu hoch wirſt ſchwüngen.\\
  Ja Barak iſt der Pfau,\\
  der feige Hochmuth bloß\\
  auf ſeinen Kopfe trägt,\\
  doch vor Schwäche ſeiner Füeßen\\
  wird Er ſich endlich doch noch ſchämen müeßen,\\
  wan Er vor mir geſtrekt\\
  in eignen Blut erſeufft wird ligen,\\
  belehrne Knechte ſeyn\\
  von meinen ſichern Siegen.
\end{lyrics}

\begin{lyrics}{Aria secunda}
  \voice[Sisera]
  Erzüttre, ertatre, erblaße Verwegner!\\
  Erfahre daß Sisera ſeye dein Gegner\\
  der ſich schon vill zu Füeßen gelegt.\\
  Und biſtu entgeiſtet zur Erden geſtrekt,\\
  ſo will ich alß Sieger den Körper beſteigen\\
  und deinen Mitknechten die Straffe anzeigen,\\[ex]
  die ihre verwegne Herzen verdient\\
  alß die ſich vergeßend nur Knechte zu ſeyn\\
  und ſich von Jabins Gebott zu befreyen,\\
  die Waffen zu ſchwingen ſich haben erkhünt.
\end{lyrics}

\begin{lyrics}{Scena quinta}
  \voice[Debora]
  Ô Herr! in deßen Wundergütte\\
  kein Grund zu forſchen iſt,\\
  befeure deines Volks Gemüthe,\\
  daß ſich durch dein Gebott zum Streitten rüſt;\\
  zum Streit, der dir beliebt,\\
  zum Entſcheid vorzurükhen,\\
  ob Jabins graußes Joch\\
  den krummen Nakhen noch\\
  von deinem Volkh ſoll fehrner drükhen.\\
  Wir kennen zwar die Schuld,\\
  die dir zu unſrer Straf\\
  die Triebe hat gereget,\\
  wir kennen aber auch die Huld,\\
  die zum Verzeyhen dich\\
  ô gueter Herr beweget.\\
  Dahero hemme doch\\
  die Wueth von unſern Plagen,\\
  und gönne unß die Luſt\\
  davor dir Dank zu ſagen.
\end{lyrics}

\begin{lyrics}{Aria tertia}
  \voice[Debora]
  Herr laß den Sieg auf jene Seithen ziegen,
  alwo daß Recht in ſeiner Waagſchall ligt.
  Dan unſer Wunſch wird ſich damit begnüegen
  ſo wie es dir Gerechtigkeit nur füegt.
  Doch würke wie es dir nur gefällt,
  wir wollen keine Rache geben,
  iſt nur daß Joch von unß getriben
  ſo bleib der Feund dir hingeſtellt.
\end{lyrics}
