\chapter{Lyrics}

\textit{Roles:} Die Barmherzigkheit (S) · Der verlohrene Sohn (A) · Der gehorſame Sohn (T) · Die Gerechtigkheit (B) · Der mitleidige Vatter (B)

\section{Actus primus}

\begin{lyrics}{Scena prima}
  \voice[Der verlohrene Sohn]
  Wo wend ich mich nun hin!\\
  die Noth hat mich umbgeben,\\
  ich bin von aller Weld\\
  verlaßen, und veracht,\\
  weil durch verkherten Sünn\\
  in meinen Wolluſtleben\\
  mein Haabſchafft, Guet und Geld\\
  nun gänzlich durchgebracht,\\
  daß Kleid der erſten Unſchuld\\
  iſt völlig ſchon zerrißen,\\
  mein Erbtheil iſt dahin,\\
  ô Sünder der ich bin!\\
  der Hunger meiner Seelen,\\
  ſo mir ins Gwißen nagt,\\
  will mich zu Todte quällen,\\
  und mir die Speiß verſagt,\\
  ſo doch mein liebſter Vatter\\
  ſein gringſten Knechten gönnet,\\
  ach daß ich doch nur bald\\
  mit ihm wär außgeſöhnet.\\
  Doch nein es iſt zu ſpatt,\\
  mein Ungehorſamb zeiget,\\
  daß ſich zu keiner Gnad\\
  deß Vatters Hertz mehr neiget.
\end{lyrics}

\begin{lyrics}{Scena secunda}
  \voice[Die Barmherzigkheit]
  Nein nein verzage nicht,\\
  du würſt noch Gnade fünden,\\
  dan deines Vatters Hertz\\
  hat allzu bittren Schmertz,\\
  daß du von ihm gewichen,\\
  bekhenne deine Schuld,\\
  werff dich zu ſeinen Füeßen,\\
  rueff nur umb Gnad und Huld,\\
  laß bittre Thränen flüeßen,\\
  ſo würſtu Gnad erlangen,\\
  Verzeühung und Pardon,\\
  Er würd dich ſelbſt umbfangen\\
  als ſeinen liebſten Sohn.
\end{lyrics}

\begin{lyrics}{Aria prima}
  \voice[Die Barmherzigkheit]
  Ô Menſch dich nicht betrüeb!\\
  dein Gott iſt ſo gearthet,\\
  daß Er nur auff dich warthet,\\
  biß du mit Reu und Schmertz\\
  bewegſt ſein mildes Hertz,\\
  dan Er iſt voll der Lieb.\\[1ex]
  Nun iſt die Gnadenzeit,\\
  wo du Verzeühung fündeſt,\\
  und Gott mit Liebe bündeſt,\\
  daß Er nicht ſtraffen kan\\
  wie Er ſchon offt gethan,\\
  drum dich zur Bueß bereüth!
\end{lyrics}

\begin{lyrics}{Scena tertia}
  \voice[Der verlohrene Sohn]
  Ach ſolte wohl mein Vatter\\
  nur gnädig känen ſeyn!\\
  da ich nicht nur allein\\
  ſein allzu treue Wahrnung\\
  ſchlechthin in Wind geſchlagen,\\
  mich von ihm abgewandet,\\
  mein Erbguet gantz verſchwendet,\\
  ſolt ich nun widerkheren,\\
  ſein Hertz auffs neu beſchweren,\\
  und Gnade alda hoffen,\\
  wo ich von ihm entloffen,\\
  diß fällt mir ſchwär zu glauben,\\
  daß Er mir ſolt erlauben\\
  ſein Angeſicht zu ſehen,\\
  ich kann es nicht geſtehen!

  \voice[Die Gerechtigkheit]
  Ja ja du urthleſt recht,\\
  verboſt-unnützer Knecht,\\
  wie ſolt dich diſer khennen,\\
  den du vorhin veracht,\\
  nur diſes wohl betracht,\\
  es hilfft hier kein Beſchönen;\\
  Wie ſehr Er dich geliebet,\\
  ſo haſtu ihm betrübet,\\
  drum gib dich nur darein,\\
  es kan nicht anderſt ſeyn,\\
  du muſt in Elend ſterben,\\
  und Ewiglich verderben.

  \voice[Der verlohrene Sohn]
  Ach ſo iſt dan gar kein Gnad!

  \voice[Die Gerechtigkheit]
  Nein nein es iſt zu ſpath.
\end{lyrics}

\begin{lyrics}{Aria secunda}
  \voice[Die Gerechtigkheit]
  Du Himmel nicht verweille\\
  mit Blitz und Donnerkeule\\
  zerſchmettre jene/diſe Brueth,\\
  die nur nach Laſter trachtet,\\
  all Tugendwerckh verachtet,\\
  ſtürtzs in die Höllengluet.\\[1ex]
  Du Erden dich erſchüttre,\\
  ergrimme, und erbüttre,\\
  verſchling diß Natterzicht,\\
  ſo alles nur vergüfftet,\\
  und großes Unheyl ſtüfftet,\\
  damit es werd gezücht.
\end{lyrics}

\begin{lyrics}{Scena quarta}
  \voice[Der verlohrene Sohn]
  O Himmel wie waß hör ich,\\
  ſo bin ich dan verſtoßen,\\
  und gäntzlich außgeſchloßen\\
  auß meines Vatters Gnad!

  \voice[Die Barmherzigkheit]
  Verzage nicht,\\
  weil dich dein Vatter liebet,\\
  ſo ſey nur unbetrüebet,\\
  leg dein Bekhäntnus ab!

  \voice[Der verlohrene Sohn]
  Diß bin ich auch wie billich\\
  von gantzen Hertzen willig,\\
  weill ich geſündigt hab.

  \voice[Die Barmherzigkheit]
  Wolan ſo mach dich auf\\
  mit kindlichen Vertrauen,\\
  thue deine Hoffnung bauen\\
  auf die Barmhertzigkheit,\\
  ſo ſchon vor dich iſt zuebereuth:\\
  dorth ſteth dein liebſter Vatter\\
  mit außgeſtreckhten Armben,\\
  will deiner ſich erbarmen\\
  auß lauter Lieb und Freud,\\
  weill du zur Bueß bereith.

  \voice[Der verlohrene Sohn]
  Vatter! Ach liebſter Vatter mein,\\
  ich hab mich ſchwär verſündigt\\
  im Himmel und auf Erden,\\
  bin auch ſchon nit mehr werth\\
  dein Kind genent zu werden,\\
  bitt wolſt mir gnädig ſeyn,\\
  mit büttereſten Schmertzen\\
  und gantz reuvollen Hertzen\\
  bekhenn ich meine Sünd,\\
  werff mich zu deinen Füeßen,\\
  ach laß mir da genüeßen\\
  daß ich Verzeyhung fünd.
\end{lyrics}

\begin{lyrics}{Aria tertia}
  \voice[Der verlohrene Sohn]
  Dan gleich wie die Waßerwellen\\
  offt auß ihren Schranckhen flüeßen\\
  und die dürre Erd begüeßen,\\
  alſo ſollen meine Augen\\
  nichts als ſcharffe Bueßelaugen\\
  gantz reumüethig hervorquellen.\\[1ex]
  Nur damit die matte Seele\\
  ſo die Sünde hat erſtückhet,\\
  werd mit ſüeßen Troſt erquickhet\\
  und zu ihren Zihl gelanget,\\
  ſo da an dem Creütze hanget\\
  ſich allein zu Gott geſelle.
\end{lyrics}

\begin{lyrics}{Scena quinta}
  \voice[Der mitleidige Vatter]
  Mein Sohn! khom ohne Scheu,\\
  wie ſehr hat mich verlanget\\
  nach deiner Reu und Bueß,\\
  nun ich bekhennen mueß,\\
  weil du ſo ſchmertzlich fleheſt,\\
  dein Müßethat geſteheſt,\\
  daß ich dir gnädig ſey.

  \voice[Der verlohrene Sohn]
  Mein Vatter!

  \voice[Der mitleidige Vatter]
  Ja ja mein Sohn.

  \voice[Der verlohrene Sohn]
  So hab ich gwiß Pardon?

  \voice[Der mitleidige Vatter]
  Nur zweiffle nicht daran.

  \voice[Die Gerechtigkheit]
  Und wie kan diß geſchehen\\
  nach ſtrenger Grechtigkheit?

  \voice[Die Barmherzigkheit]
  Dieweil er Bueß noch würkhet\\
  alhier in diſer Zeit.

  \voice[Die Gerechtigkheit]
  Er mueß geſtürtzet werden\\
  ins Ewige Verderben.

  \voice[Die Barmherzigkheit]
  Er kan noch als ein Sohn\\
  deß Vatters Reich erwerben.

  \voice[Die Gerechtigkheit]
  Die Boßheit diſes Sünders\\
  iſt gar unendlich groß.

  \voice[Die Barmherzigkheit]
  Die Bueß ſo er gewürckhet,\\
  macht ihm von Sünden loß.

  \voice[Die Gerechtigkheit]
  Er ſterbe in ſein Ruin.

  \voice[Die Barmherzigkheit]
  Er lebe noch fehrnerhin.
\end{lyrics}

\begin{lyrics}{Aria quarte}
  \voice[Der mitleidige Vatter]
  Eß lebe dan mein Kind,\\
  ich nehm auf mich die Sünd,\\
  und ich will vor ihm ſterben\\
  daß er nicht ſolt verderben,\\
  wan ſchon der ärgſte Todt,\\
  ja alle Angſt und Noth\\
  vor mich bereithet wär,\\
  die acht ich gar nicht mehr:\\[1ex]
  Diß macht die Lieb allein,\\
  ſo mich ſchon feſt gebunden,\\
  auf daß die Grechtigkheit\\
  an mir erfüllet werd,\\
  drum acht ich keine Peyn,\\
  die Forcht iſt gantz verſchwunden,\\
  weil die Barmherzigkheit\\
  dem Sünder Gnad beſchert.
\end{lyrics}

\begin{lyrics}{Scena sexta}
  \voice[Der verlohrene Sohn]
  Ach wohl ein lieber Vatter\\
  der ſeine aigne Seele\\
  vor meine Seele ſezt,\\
  ſich ſelbſt auch dar will geben\\
  vor mein boßhafftes Leben,\\
  damit ich endlich werde\\
  mit ſeinen Bluet benezt.

  \voice[Der mitleidige Vatter]
  Diß hat der höchſte Rath\\
  von Anbegin beſchloßen,\\
  weil ſonſt der ſündig Menſch\\
  auf ewig wär verſtoßen;\\
  drumb ſoſt daß Kleid der Unſchuld\\
  aufs neue überkhommen,\\
  den Goldring ſchenkh ich dir\\
  zum Zeichen meiner Liebe,\\
  ſey nur hinforth volkhommen,\\
  und dich der Tugend yebe.

  \voice[Der gehorſame Sohn]
  Wie kombts mein liebſter Vatter,\\
  daß du diſen Bößwicht\\
  den Verſchwender viller Gnaden\\
  mit Reichtumb deiner Güetter\\
  aufs neue wilſt beladen,\\
  ja gar dein aignes Leben\\
  vor ihm auch dar wilſt geben;\\
  war ich nicht alzeit ghorſamb,\\
  und ſtäts an deiner Seithen,\\
  wie laſt dich von der Lieb\\
  nur gar ſo ſehr verleithen,\\
  indeme du dergleichen\\
  an mir niemahls gethan,\\
  wie komts, daß ich muß weichen\\
  dem ungerathnen Sohn?

  \voice[Der mitleidige Vatter]
  Mein Sohn gib dich zufriden,\\
  dir gſchiecht kein Unrecht nicht!\\
  dann daß du recht gethan,\\
  diß war ja deine Schuld und Pflicht,\\
  biſtu darum ſchadhafftig,\\
  daß ich ſo güetig bin,\\
  hab ich dan nicht die Macht,\\
  wie redeſt du ſo khün?\\
  Kann ich wohl diſen haßen,\\
  der umb Verzeyhung bitt,\\
  und ſolte ihm verlaßen,\\
  nein nein daß glaube nit,\\
  der Sünder ſo da büeßet,\\
  iſt auch ſchon gäntzlich mein,\\
  drum werde dorth die lezte\\
  wohl gar die Erſte ſeyn.

  \voice[Der gehorſame Sohn]
  Verzeyhe liebſter Vatter!\\
  Ich geb mich willig drein.
\end{lyrics}

\begin{lyrics}{Aria quinta}
  \voice[Der gehorſame Sohn]
  Mach ſich nur keiner grecht!\\
  dan wer da ſteth, der habe acht,\\
  daß Er nicht werd zum Fahl gebracht,\\
  ob er gleich ſchon alle Tugend\\
  von ſeiner erſten blüehenden Jugend\\
  gewürckhet hat/hätt mit Bueß und Reu,\\
  ſoll er ſich doch bekhennen frey\\
  nur als ein unnützer Knecht.\\[1ex]
  Drum Sünder habe acht!\\
  weill deine gantze Lebenszeit\\
  nichts anders als ein purer Streit,\\
  ſo würckh dein Heil und züttre,\\
  nur Gott nicht mehr erbüttre,\\
  der dich zugleich mit Leib und Seel\\
  kan ſtürtzen in Abgrund der Höll,\\
  nur diſes wohl betracht!
\end{lyrics}
